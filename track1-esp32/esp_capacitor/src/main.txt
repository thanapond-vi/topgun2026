// ESP32 CAN R01 - pF meter (PlatformIO)
//
// Principle: Cx is charged from IO16 through R1 (1 MΩ). An LM393 compares the
// Cx voltage with Vref (VCC/2). The time until the comparator output goes high
// gives  t = R1 * Cx * ln(2)  ->  Cx = t / (R1 * ln 2).
//
// Serial commands (9600 baud, line terminated or not):
//   m          single measurement
//   a          toggle continuous measurement
//   z          zero: store stray capacitance (probes open, no Cx)
//   k <pF>     calibrate scale with a known capacitor, e.g. "k 100"
//   r          reset zero + calibration to defaults
//   ?          help / show settings
// SW2 button also triggers a single measurement.
// External switch (PIN_EXT to GND): short press = measure, hold 1.5 s = zero.
//
// Zero and calibration are stored in flash (NVS) and survive reboots.

#include <Arduino.h>
#include <Preferences.h>
#include <math.h>
#include <stdlib.h>
#include "soc/gpio_reg.h"

// ================= ESP32 CAN R01 pin map =================
constexpr int PIN_CHG     = 16;   // IO16 -> R1 (1 MΩ) -> Cx   (must be < 32 for fast GPIO path)
constexpr int PIN_CMP     = 17;   // IO17 <- LM393 OUT, 2k pull-up to 3.3V
constexpr int PIN_BTN     = 32;   // SW2 on board (1k external pull)
constexpr int PIN_LED     = 2;    // typical ESP32 built-in LED
constexpr int BTN_PRESSED = LOW;  // LOW if 1k is pull-up, HIGH if pull-down
// External switch: wire between PIN_EXT and GND (internal pull-up is used).
//   short press = measure, hold >= LONG_PRESS_MS = zero
constexpr int      PIN_EXT        = 33;   // any free GPIO with internal pull-up (0-33), change if needed
constexpr int      EXT_PRESSED    = LOW;  // LOW = switch to GND
constexpr uint32_t LONG_PRESS_MS  = 1500;
// NOTE: IO16/IO17 are used by PSRAM on WROVER modules. Use a WROOM module.

static_assert(PIN_CHG < 32, "PIN_CHG must be a GPIO below 32");

// ================= Circuit values =================
constexpr double R1   = 1.0e6;    // Ω
constexpr double R2   = 10000.0;  // Ω (3.3V -> V-)
constexpr double R3   = 10000.0;  // Ω (V- -> GND)
constexpr double VCC  = 3.3;
constexpr double DEFAULT_VREF = VCC * R3 / (R2 + R3);   // 1.65 V
constexpr double FIXED_ZERO_PF = 22.15;
constexpr double CALIBRATION_REFERENCE_PF = 20.0;

// ================= Measurement settings =================
constexpr int      SAMPLES        = 9;      // readings per measurement (median-trimmed mean)
constexpr uint32_t DISCHARGE_MS   = 10;
constexpr uint32_t TIMEOUT_MS     = 20;     // max charge time (~29 nF with 1 MΩ)
constexpr uint32_t CONT_PERIOD_MS = 500;

// ================= State =================
enum class Status : uint8_t { OK, NOT_DISCHARGED, TIMEOUT };

static Preferences prefs;
static double   strayPF   = 0.0;    // additional measured zero offset in pF
static double   scale     = 1.0;   // calibration factor
static double   vref      = DEFAULT_VREF;
static uint32_t cpuHz     = 240000000UL;
static bool     continuous = false;
static uint32_t lastCont   = 0;
static uint32_t lastLedOn  = 0;
static bool     ledOn      = false;
static portMUX_TYPE mux = portMUX_INITIALIZER_UNLOCKED;

// ---- Fast GPIO access ----
static inline __attribute__((always_inline)) void IRAM_ATTR chgHigh() {
  REG_WRITE(GPIO_OUT_W1TS_REG, 1UL << PIN_CHG);
}
static inline __attribute__((always_inline)) bool IRAM_ATTR cmpHigh() {
  if (PIN_CMP < 32) return REG_READ(GPIO_IN_REG)  & (1UL << PIN_CMP);
  else              return REG_READ(GPIO_IN1_REG) & (1UL << (PIN_CMP >= 32 ? PIN_CMP - 32 : 0));
}

// ---- One raw charge-time measurement (no zero / calibration applied) ----
static Status measureRawOnce(double &pF) {
  digitalWrite(PIN_CHG, LOW);                  // discharge Cx
  delay(DISCHARGE_MS);
  if (digitalRead(PIN_CMP)) return Status::NOT_DISCHARGED;  // OUT must be LOW

  const uint32_t timeoutCycles = (cpuHz / 1000UL) * TIMEOUT_MS;
  uint32_t t0, t1;

  portENTER_CRITICAL(&mux);
  t0 = ESP.getCycleCount();
  chgHigh();
  do { t1 = ESP.getCycleCount(); }
  while (!cmpHigh() && (t1 - t0) < timeoutCycles);
  portEXIT_CRITICAL(&mux);

  digitalWrite(PIN_CHG, LOW);
  if ((t1 - t0) >= timeoutCycles) return Status::TIMEOUT;

  const double t = (double)(t1 - t0) / cpuHz;  // seconds
  pF = -t / (R1 * log(1.0 - vref / VCC)) * 1e12;
  return Status::OK;
}

static int cmpDouble(const void *a, const void *b) {
  double x = *(const double *)a, y = *(const double *)b;
  return (x > y) - (x < y);
}

// ---- Averaged raw measurement: sort, drop outer quarters, average the middle ----
static Status measureRaw(double &pF) {
  double v[SAMPLES];
  for (int i = 0; i < SAMPLES; i++) {
    Status s = measureRawOnce(v[i]);
    if (s != Status::OK) return s;
  }
  qsort(v, SAMPLES, sizeof(double), cmpDouble);
  const int lo = SAMPLES / 4, hi = SAMPLES - SAMPLES / 4;
  double sum = 0;
  for (int i = lo; i < hi; i++) sum += v[i];
  pF = sum / (hi - lo);
  return Status::OK;
}

// ---- Final measurement with zero + calibration applied ----
static Status measurePF(double &pF) {
  double raw;
  Status s = measureRaw(raw);
  if (s != Status::OK) return s;
  pF = (raw - strayPF - FIXED_ZERO_PF) * scale;
  return Status::OK;
}

// ---- Persistence ----
static void loadSettings() {
  prefs.begin("pfmeter", true);
  strayPF = prefs.getDouble("stray", 0.0);
  scale   = prefs.getDouble("scale", 1.0);
  vref    = prefs.getDouble("vref", DEFAULT_VREF);
  prefs.end();
  if (vref <= 0.0 || vref >= VCC) vref = DEFAULT_VREF;
}
static void saveSettings() {
  prefs.begin("pfmeter", false);
  prefs.putDouble("stray", strayPF);
  prefs.putDouble("scale", scale);
  prefs.putDouble("vref", vref);
  prefs.end();
}

// ---- Output ----
static void printStatusError(Status s) {
  if (s == Status::NOT_DISCHARGED)
    Serial.println("Error: comparator stuck HIGH after discharge (check wiring / Vref divider / Cx leakage)");
  else if (s == Status::TIMEOUT)
    Serial.println("Error: timeout (Cx too large >~29 nF, or no charge path / comparator not switching)");
}

static void printValue(double pF) {
  if (pF > -0.5 && pF < 0.0) pF = 0.0;         // hide tiny negative noise
  if (fabs(pF) >= 1000.0) Serial.printf("C = %.3f nF\n", pF / 1000.0);
  else                    Serial.printf("C = %.2f pF\n", pF);
}

static void doMeasure() {
  double pF;
  Status s = measurePF(pF);
  if (s == Status::OK) printValue(pF);
  else                 printStatusError(s);
}

static void doZero() {
  double raw;
  Status s = measureRaw(raw);
  if (s != Status::OK) { printStatusError(s); return; }
  strayPF = raw - FIXED_ZERO_PF;
  saveSettings();
  Serial.printf("Zero set: %.2f pF (saved)\n", strayPF);
}

static void printHelp() {
  Serial.println("ESP32 CAN R01 - pF meter");
  Serial.printf("Vref = %.3f V | zero = %.2f pF | scale = %.4f | CPU = %u MHz\n",
                vref, strayPF, scale, (unsigned)(cpuHz / 1000000UL));
  Serial.println("Commands: m = measure | a = continuous | z = zero | k = calibrate with 20 pF | v <V> = set Vref | r = reset | ? = help");
  Serial.println("SW2 = measure | external switch: short = measure, hold = zero");
}

// ---- Command handling ----
static void handleCommand(const char *line) {
  while (*line == ' ') line++;
  if (!*line) return;

  switch (line[0]) {
    case 'm': doMeasure(); break;

    case 'a':
      continuous = !continuous;
      Serial.println(continuous ? "Continuous: ON" : "Continuous: OFF");
      break;

    case 'z': doZero(); break;

    case 'v': {
      double enteredVref = atof(line + 1);
      if (enteredVref <= 0.0 || enteredVref >= VCC) {
        Serial.printf("Usage: v <voltage from 0 to %.1f V>\n", VCC);
        break;
      }
      vref = enteredVref;
      saveSettings();
      Serial.printf("Vref set: %.3f V (saved)\n", vref);
      break;
    }

    case 'k': {
      double raw;
      Status s = measureRaw(raw);
      if (s != Status::OK) { printStatusError(s); break; }
      double net = raw - strayPF - FIXED_ZERO_PF;
      if (net <= 1.0) { Serial.println("Error: reading too small - connect the known capacitor first"); break; }
      scale = CALIBRATION_REFERENCE_PF / net;
      saveSettings();
      Serial.printf("Calibrated with %.2f pF: scale = %.4f (saved)\n",
                    CALIBRATION_REFERENCE_PF, scale);
      break;
    }

    case 'r':
      strayPF = 0.0; scale = 1.0; vref = DEFAULT_VREF;
      saveSettings();
      Serial.println("Zero and calibration reset");
      break;

    case '?': case 'h': printHelp(); break;

    default: Serial.println("Unknown command, '?' for help"); break;
  }
}

static void pollSerial() {
  static char buf[32];
  static uint8_t n = 0;
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (n) { buf[n] = 0; handleCommand(buf); n = 0; }
    } else if (n < sizeof(buf) - 1) {
      buf[n++] = c;
      // single-letter commands without Enter (not 'k', which needs a value)
      if (n == 1 && (c == 'm' || c == 'a' || c == 'z' || c == 'r' || c == '?')) {
        buf[n] = 0; handleCommand(buf); n = 0;
      }
    }
  }
}

// ---- Button ----
static bool buttonPressed() {
  static int lastReading = !BTN_PRESSED;
  static int stableState = !BTN_PRESSED;
  static uint32_t lastDebounce = 0;

  const int reading = digitalRead(PIN_BTN);
  if (reading != lastReading) {
    lastDebounce = millis();
    lastReading = reading;
  }

  if (millis() - lastDebounce < 30) return false;
  if (stableState == reading) return false;

  stableState = reading;
  return stableState == BTN_PRESSED;
}

// ---- External switch: short press = measure, long press = zero ----
enum class Press : uint8_t { NONE, SHORT, LONG };

static Press extSwitch() {
  if (digitalRead(PIN_EXT) != EXT_PRESSED) return Press::NONE;
  delay(30);                                   // debounce
  if (digitalRead(PIN_EXT) != EXT_PRESSED) return Press::NONE;

  const uint32_t t0 = millis();
  while (digitalRead(PIN_EXT) == EXT_PRESSED && millis() - t0 < LONG_PRESS_MS) delay(10);

  const bool isLong = digitalRead(PIN_EXT) == EXT_PRESSED;
  while (digitalRead(PIN_EXT) == EXT_PRESSED) delay(10);   // wait for release
  delay(30);
  return isLong ? Press::LONG : Press::SHORT;
}

void setup() {
  Serial.begin(9600);
  pinMode(PIN_CHG, OUTPUT);
  digitalWrite(PIN_CHG, LOW);
  pinMode(PIN_CMP, INPUT);                     // external 2 kΩ pull-up
  pinMode(PIN_BTN, INPUT);                     // external 1 kΩ on board
  pinMode(PIN_LED, OUTPUT);
  digitalWrite(PIN_LED, LOW);
  pinMode(PIN_EXT, INPUT_PULLUP);              // external switch to GND
  cpuHz = ESP.getCpuFreqMHz() * 1000000UL;
  loadSettings();
  delay(200);
  printHelp();
}

void loop() {
  pollSerial();

  if (ledOn && millis() - lastLedOn >= 200) {
    digitalWrite(PIN_LED, LOW);
    ledOn = false;
  }

  if (buttonPressed()) {
    digitalWrite(PIN_LED, HIGH);
    lastLedOn = millis();
    ledOn = true;
    doMeasure();
  }

  switch (extSwitch()) {
    case Press::SHORT: doMeasure(); break;
    case Press::LONG:  doZero();    break;
    default: break;
  }

  if (continuous && millis() - lastCont >= CONT_PERIOD_MS) {
    lastCont = millis();
    doMeasure();
  }
}
